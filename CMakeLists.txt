cmake_minimum_required(VERSION 3.31)

project(PalmierProWindowsContracts LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "The Palmier Pro Windows probe requires Windows.")
endif()

if(NOT MSVC)
    message(FATAL_ERROR "The Palmier Pro Windows probe requires MSVC.")
endif()

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Palmier Pro Windows probe requires an x64 toolchain.")
endif()

include(CTest)
add_subdirectory(core/audio)
add_subdirectory(core/project)
add_subdirectory(core/project-session)
add_subdirectory(core/project-runtime)
add_subdirectory(core/render)
add_subdirectory(core/project-render)
add_subdirectory(windows/render-d3d11)
add_subdirectory(windows/contract-probe)
add_subdirectory(windows/mcp-http)
add_subdirectory(windows/audio-wasapi)

option(PALMIER_ENABLE_FFMPEG_PROTOTYPE "Build the FFmpeg media prototype" OFF)
if(PALMIER_ENABLE_FFMPEG_PROTOTYPE)
    add_subdirectory(windows/media-ffmpeg)
    add_subdirectory(windows/media-render)
    add_subdirectory(windows/export-ffmpeg)
    add_subdirectory(windows/media-session)
    add_subdirectory(windows/preview-session)
endif()

option(PALMIER_ENABLE_QT_SHELL "Build the optional Qt/QML read-only shell" OFF)
if(PALMIER_ENABLE_QT_SHELL)
    if(NOT PALMIER_ENABLE_FFMPEG_PROTOTYPE)
        message(FATAL_ERROR "The Qt shell requires the FFmpeg preview prototype")
    endif()
    add_subdirectory(windows/app)
endif()
